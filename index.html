Piezas por operaria
⚙️ Guía de configuración con Google Sheets
Sigue estos 4 pasos para conectar la app con tu hoja de cálculo
Paso 1 — Crea el Google Sheet
Crea una hoja llamada Produccion (sin tilde) con estas columnas en la fila 1, exactamente en este orden:
A: ID B: Marca_Temporal C: Fecha_Produccion D: Operaria E: Usuario F: Referencia G: Cantidad H: Operacion I: Observaciones J: Estado
Paso 2 — Crea el Apps Script
En tu Google Sheet ve a Extensiones → Apps Script, borra el código que aparece por defecto y pega el siguiente: 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Produccion");
  var p = JSON.parse(decodeURIComponent(e.parameter.payload || '{}'));
  var callback = e.parameter.callback || '';
  var res = {};

  if (p.accion === 'registrar') {
    var id = Utilities.getUuid().slice(0, 8);
    sheet.appendRow([
      id,
      p.marca_temporal,
      p.fecha,
      p.operaria,
      p.usuario,
      p.referencia,
      p.cantidad,
      p.operacion,
      p.observaciones,
      'Pendiente'
    ]);
    res = { ok: true };

  } else if (p.accion === 'obtener') {
    var data = sheet.getDataRange().getValues();
    var registros = [];
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][4]) === String(p.usuario))
        registros.push({
          id: data[i][0], marca_temporal: data[i][1],
          fecha: data[i][2], operaria: data[i][3],
          usuario: data[i][4], referencia: data[i][5],
          cantidad: data[i][6], operacion: data[i][7],
          observaciones: data[i][8], estado: data[i][9]
        });
    }
    res = { registros: registros };

  } else if (p.accion === 'obtenerTodos') {
    var data = sheet.getDataRange().getValues();
    var registros = [];
    for (var i = 1; i < data.length; i++) {
      registros.push({
        id: data[i][0], marca_temporal: data[i][1],
        fecha: data[i][2], operaria: data[i][3],
        usuario: data[i][4], referencia: data[i][5],
        cantidad: data[i][6], operacion: data[i][7],
        observaciones: data[i][8], estado: data[i][9]
      });
    }
    res = { registros: registros };

  } else if (p.accion === 'actualizarEstado') {
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(p.id)) {
        sheet.getRange(i + 1, 10).setValue(p.estado);
        break;
      }
    }
    res = { ok: true };
  }

  var json = JSON.stringify(res);
  if (callback) {
    return ContentService
      .createTextOutput(callback + '(' + json + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}
Paso 3 — Publica como Web App
En el editor de Apps Script: Implementar → Nueva implementación
• Tipo: Aplicación web
• Ejecutar como: Yo (tu cuenta Google)
• Quién tiene acceso: Cualquier usuario
Haz clic en Implementar y autoriza los permisos cuando te los pida.
Paso 4 — Pega la URL en el HTML
Copia la URL que te da Google (termina en /exec).
Abre este archivo HTML en un editor de texto, busca la línea:
const SCRIPT_URL = 'TU_SCRIPT_URL_AQUI';
y reemplaza TU_SCRIPT_URL_AQUI con tu URL. Guarda y vuelve a abrir el HTML.
